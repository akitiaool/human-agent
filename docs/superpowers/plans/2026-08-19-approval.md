# 危险命令人工授权（Gemini 桥 v3）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 桥服务遇危险命令不再自动拒绝，而是向扩展下发审批请求；用户在网页弹窗授权（允许/拒绝/超时/自动授权开关），桥服务按审批结果执行或拒绝。

**Architecture:** 审批请求经现有 `GET /task` 长轮询通道下发（响应改为 `{type:"message"|"approval"}` 联合类型），新增 `POST /approval` 回传逐条裁决。桥服务在 `#process` 中预筛摘出危险命令 → 批量审批（兜底超时 600s 可注入）→ 按原 calls 顺序执行/拒绝。扩展侧自建暗色弹窗（倒计时 + 逐条按钮 + 全部允许），面板新增「自动授权」开关与「审批超时」配置（插件层超时先行，默认 120s）。

**Tech Stack:** TypeScript strict + ESM（Node ≥20）、node:http、vitest 4（TDD）、chrome MV3 扩展（无构建，原生 JS + storage.sync）、pnpm 11。

**Spec:** `docs/superpowers/specs/2026-08-19-approval-design.md`（本计划从其论证；v2 桥语义见 `docs/superpowers/specs/2026-08-19-gemini-bridge-design.md`）

```ts
type OutboxBody =
  | { type: "message"; message: string }
  | { type: "approval"; id: number; calls: Array<{ name: string; arguments: Record<string, unknown> }> };

/** 长轮询取一条出队消息；无消息时服务端 25s 后 204（测试不等待该路径）。 */
async function getMessage(port: number): Promise<OutboxBody | null> {
  const res = await fetch(`http://127.0.0.1:${port}/task`);
  if (res.status === 204) return null;
  return (await res.json()) as OutboxBody;
}

/** 取回填文本消息（非审批类型时抛错）。 */
async function getMessageText(port: number): Promise<string> {
  const body = await getMessage(port);
  if (body === null) throw new Error("expected message, got 204");
  if (body.type !== "message") throw new Error(`expected message, got ${body.type}`);
  return body.message;
}

/** 提交审批结果，返回 HTTP 状态码。 */
async function postApproval(port: number, id: number, allowed: boolean[]): Promise<number> {
  const res = await fetch(`http://127.0.0.1:${port}/approval`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ id, allowed }),
  });
  await res.json().catch(() => undefined);
  return res.status;
}
```

(8) 新增审批方法（`#pushMessage` 之后）：

```ts
  /** 审批请求入队并返回裁决 promise；兜底超时（approvalTimeoutMs）全部按拒绝。 */
  #requestApproval(calls: readonly ToolCall[]): { id: number; promise: Promise<ApprovalDecision> } {
    const id = this.#nextApprovalId++;
    let settle!: (decision: ApprovalDecision) => void;
    const timer = setTimeout(() => {
      if (this.#approvals.delete(id)) settle({ allowed: calls.map(() => false), timedOut: true });
    }, this.#approvalTimeoutMs);
    const promise = new Promise<ApprovalDecision>((resolve) => {
      settle = (decision) => {
        clearTimeout(timer);
        resolve(decision);
      };
    });
    this.#approvals.set(id, { settle, count: calls.length });
    this.#outbox.push({ type: "approval", id, calls: [...calls] });
    this.#logger?.(`审批请求已入队（id ${id}，${calls.length} 条危险命令）`);
    this.#wakeWaiter();
    return { id, promise };
  }

  /** 收扩展审批结果：已知且未处理的 id → 逐条放行；未知 id（已超时/已处理）→ 404。 */
  #handleApprovalRequest(req: http.IncomingMessage, res: http.ServerResponse): void {
    readJsonBody(req).then(
      (body) => {
        const record = (typeof body === "object" && body !== null ? body : {}) as Record<string, unknown>;
        const id = typeof record.id === "number" ? record.id : NaN;
        const allowed = (Array.isArray(record.allowed) ? record.allowed : []).map((v) => v === true);
        const entry = this.#approvals.get(id);
        if (entry === undefined || allowed.length === 0) {
          res.writeHead(404, { "Content-Type": "application/json" });
          res.end(JSON.stringify({ error: "unknown approval" }));
          return;
        }
        this.#approvals.delete(id);
        entry.settle({ allowed, timedOut: false });
        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ ok: true }));
      },
      () => {
        res.writeHead(400, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ error: "请求体必须为合法 JSON" }));
      },
    );
  }
```

(9) 替换 `#process` 整体：

```ts
  /** 单批流水线：预筛摘出危险命令 → 反思裁决其余 → 批量审批 → 按原 calls 顺序执行/拒绝 → 回填。 */
  async #process(calls: readonly ToolCall[]): Promise<void> {
    if (this.#signal.aborted) return; // 全局中断：丢弃批次
    const batchSignal =
      this.#timeoutMs > 0
        ? AbortSignal.any([this.#signal, AbortSignal.timeout(this.#timeoutMs)])
        : this.#signal;

    // 预筛：参数合法且鼻子判定危险的 terminal 摘出；参数非法的危险命令留给 Reflection 以 INVALID_ARGS 拒绝
    const dangerous: ToolCall[] = [];
    const rest: ToolCall[] = [];
    for (const call of calls) {
      if (call.name === "terminal") {
        const parsed = schemas.terminal.safeParse(call.arguments);
        if (parsed.success) {
          const { command } = parsed.data as { command: string };
          if (this.#nose.checkRisk(command) === "DANGEROUS") {
            dangerous.push(call);
            continue;
          }
        }
      }
      rest.push(call);
    }

    // 批量审批：一条请求含全部危险命令；批中止时作废审批条目
    let decision: ApprovalDecision | undefined;
    if (dangerous.length > 0) {
      const { id, promise } = this.#requestApproval(dangerous);
      try {
        decision = await Promise.race([promise, abortSignalPromise(batchSignal)]);
      } catch {
        this.#approvals.delete(id);
        if (this.#signal.aborted) return; // 全局中断：丢弃批次
        this.#pushMessage(`⚠️ 工具执行超时（超过 ${this.#timeoutMs}ms），已终止`);
        return;
      }
    }

    const verdict = await this.#reflection.reflect(rest);
    const dangerSet = new Set(dangerous);
    const allowedSet = new Set(verdict.allowed);
    const blockedMap = new Map(verdict.blocked.map((item) => [item.call, item.result] as const));
    const decisions = decision?.allowed ?? [];
    const lines: string[] = [];
    let index = 0;
    let dangerIndex = 0;
    for (const call of calls) {
      // 原 calls 顺序执行与输出（执行顺序 = 原顺序，逐条串行）
      let toRun: ToolCall | undefined;
      let blockedResult: ActionResult<unknown> | undefined;
      if (dangerSet.has(call)) {
        const allowed = decisions[dangerIndex++] ?? false;
        if (allowed) {
          toRun = call;
        } else {
          blockedResult = fail("USER_DENIED", decision?.timedOut === true ? "审批超时，未执行" : "用户未授权，未执行");
        }
      } else if (allowedSet.has(call)) {
        toRun = call;
      } else {
        blockedResult = blockedMap.get(call);
      }
      if (toRun !== undefined) {
        index++;
        if (batchSignal.aborted) break; // 批中止：停止执行
        let result: ActionResult<unknown>;
        try {
          result = await Promise.race([
            executeTool(this.#organs, toRun.name, toRun.arguments, batchSignal),
            abortSignalPromise(batchSignal),
          ]);
        } catch {
          break; // 批 signal 已中止（执行未在超时前返回）
        }
        lines.push(serializeResultLine(index, toRun, result));
      } else if (blockedResult !== undefined) {
        index++;
        lines.push(serializeBlockedLine(index, call, blockedResult));
      }
    }
    if (this.#signal.aborted) return; // 全局中断：丢弃批次不排队
    if (batchSignal.aborted) {
      this.#pushMessage(`⚠️ 工具执行超时（超过 ${this.#timeoutMs}ms），已终止`);
      return;
    }
    this.#pushMessage(serializeFeedback(lines));
  }
```

```js
// ---------- 审批：危险命令授权（弹窗/自动授权）→ POST /approval ----------

async function postApproval(id, allowed) {
  try {
    const resp = await chrome.runtime.sendMessage({ type: "approve", id, allowed, baseUrl: settings.baseUrl });
    if (resp?.ok === true) return true;
    if (resp?.status === 404) {
      log("审批请求已过期（桥服务已按超时拒绝），跳过");
      return true; // 桥已兜底处理并回填拒绝行，流程继续
    }
    noteBridgeFailure(resp?.error ?? "POST /approval 失败");
    return false;
  } catch (err) {
    noteBridgeFailure(err instanceof Error ? err.message : String(err));
    return false;
  }
}

async function handleApproval(item) {
  if (settings.autoApprove) {
    log(`自动授权已开启：静默放行 ${item.calls.length} 条危险命令`);
    if (!(await postApproval(item.id, item.calls.map(() => true)))) setStatus("错误");
    return;
  }
  setStatus("等待授权");
  log(`收到审批请求（${item.calls.length} 条危险命令），等待授权…`);
  const decision = await showApprovalDialog(item);
  if (!(await postApproval(item.id, decision.allowed))) setStatus("错误");
  setStatus("待命");
}

// 审批弹窗：遮罩 + 居中卡片；逐条 [允许][拒绝] + 全部允许/全部拒绝；
// 顶部倒计时按面板审批超时配置，到点自动全部拒绝。
function showApprovalDialog(item) {
  return new Promise((resolve) => {
    const overlay = document.createElement("div");
    overlay.className = "ha-approval-overlay";
    const box = document.createElement("div");
    box.className = "ha-approval-box";
    const secondsLeft = { value: settings.approvalTimeoutSec };
    const rows = item.calls
      .map(
        (call, i) => `
      <div class="ha-approval-row" data-i="${i}">
        <div class="ha-approval-cmd">${escapeHtml(call.arguments?.command ?? JSON.stringify(call.arguments))}</div>
        <div class="ha-approval-hint">该命令可能删除或覆盖数据，请确认后授权</div>
        <div class="ha-approval-btns">
          <button class="ha-allow" data-i="${i}">允许</button>
          <button class="ha-deny" data-i="${i}">拒绝</button>
        </div>
      </div>`,
      )
      .join("");
    box.innerHTML = `
      <div class="ha-approval-title">危险命令授权 <span class="ha-approval-count" id="ha-approval-count"></span></div>
      ${rows}
      <div class="ha-approval-foot">
        <button class="ha-allow-all" id="ha-approval-allow-all">全部允许</button>
        <button class="ha-deny-all" id="ha-approval-deny-all">全部拒绝</button>
      </div>`;
    overlay.appendChild(box);
    document.documentElement.appendChild(overlay);

    const decisions = item.calls.map(() => false);
    const countEl = box.querySelector("#ha-approval-count");
    const finish = (allowed) => {
      clearInterval(timer);
      overlay.remove();
      resolve({ allowed });
    };
    const timer = setInterval(() => {
      secondsLeft.value -= 1;
      countEl.textContent = `${secondsLeft.value}s 后自动拒绝`;
      if (secondsLeft.value <= 0) finish(item.calls.map(() => false)); // 超时：全部拒绝
    }, 1000);
    countEl.textContent = `${secondsLeft.value}s 后自动拒绝`;

    box.addEventListener("click", (e) => {
      const btn = e.target.closest("button");
      if (!btn) return;
      const i = Number(btn.dataset.i);
      if (btn.classList.contains("ha-allow")) {
        decisions[i] = true;
        finish(decisions);
      } else if (btn.classList.contains("ha-deny")) {
        decisions[i] = false;
        finish(decisions);
      } else if (btn.id === "ha-approval-allow-all") {
        finish(item.calls.map(() => true));
      } else if (btn.id === "ha-approval-deny-all") {
        finish(item.calls.map(() => false));
      }
    });
  });
}
```
